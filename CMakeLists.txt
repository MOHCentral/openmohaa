#[=======================================================================[.rst:
OpenMoHAA GDExtension — CMake Build (Windows)
----------------------------------------------
Builds the openmohaa GDExtension shared library using CMake + MSVC.
This sidesteps a Python RecursionError that occurs in SCons with 1400+
source files on Windows CI runners.

Other platforms (Linux, macOS, Web, Android) continue to use SCons.

Usage (from repo root):
    cmake -B build-win -S openmohaa ^
          -DGODOTCPP_TARGET=template_debug ^
          -DGODOTCPP_DEV_BUILD=ON ^
          -DGODOTCPP_DISABLE_EXCEPTIONS=OFF
    cmake --build build-win --config Debug
]=======================================================================]
cmake_minimum_required(VERSION 3.17)

# Set godot-cpp options as cache variables BEFORE add_subdirectory.
# godotcpp_options() uses non-FORCE CACHE sets, so pre-existing cache
# values (from -D flags or these defaults) take precedence.
if(NOT DEFINED CACHE{GODOTCPP_TARGET})
    set(GODOTCPP_TARGET "template_debug" CACHE STRING "Godot build target")
endif()
if(NOT DEFINED CACHE{GODOTCPP_DEV_BUILD})
    set(GODOTCPP_DEV_BUILD ON CACHE BOOL "Developer build")
endif()
if(NOT DEFINED CACHE{GODOTCPP_DISABLE_EXCEPTIONS})
    set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "Disable exceptions (must be OFF for engine)")
endif()
if(NOT DEFINED CACHE{GODOTCPP_SYMBOL_VISIBILITY})
    set(GODOTCPP_SYMBOL_VISIBILITY "visible" CACHE STRING "Symbol visibility")
endif()

project(openmohaa
    VERSION 1.0
    DESCRIPTION "OpenMoHAA GDExtension"
    LANGUAGES C CXX
)

# Build godot-cpp as a static library.
add_subdirectory(../godot-cpp ${CMAKE_CURRENT_BINARY_DIR}/godot-cpp)

# ──────────────────────────────────────────────
#  Exclusion filters — matches SConstruct exactly
# ──────────────────────────────────────────────

set(_EXCLUDED "")
foreach(SRC ${ALL_SOURCES})
    # Normalise to forward slashes and lowercase for case-insensitive matching
    # (matches SConstruct's fp = str(filepath).lower() approach)
    string(REPLACE "\\" "/" _SN "${SRC}")
    string(TOLOWER "${_SN}" _SN)

    set(_EXCL FALSE)

    # GameSpy common/linux/ — included via #include from platform wrappers
    if(_SN MATCHES "/code/gamespy/common/linux/")
        set(_EXCL TRUE)
    endif()

    # Old parser duplicates (we use the generated/ versions)
    if(_SN MATCHES "/code/parser/lex\\.yy\\.cpp$" OR _SN MATCHES "/code/parser/y\\.tab\\.cpp$")
        set(_EXCL TRUE)
    endif()

    # GameSpy platform/sample directories
    if(NOT _EXCL)
        foreach(_PAT
            "/mfc/" "/win32/"
            "/gp_stats/" "/laddertrack/" "/multitrack/" "/track/" "/chatty/"
            "/sbmfcsample/" "/scracesample/" "/peerlobby/" "/cdkeygen/"
            "/gt2proxy/" "/voice2bench/" "/gt2action/" "/gt2nat/" "/nitro/"
            "/revolution/" "/ps2/" "/ps3/" "/psp/" "/macosx/" "/xbox/"
            "/x360/" "/common_light/" "/gt2hostmig/" "/chatc/"
            "/gpstress/" "/peerc/" "/qr2csample/" "/fpupdate/"
            "/webservices/" "/peer/" "/voice2/"
            "/ghttp/ghttpmfc/"
            "/gp/"
            "/sc/"
            "/sake/"
        )
            if(_SN MATCHES "${_PAT}")
                set(_EXCL TRUE)
                break()
            endif()
        endforeach()
    endif()

    # Test directories and test_ files
    if(_SN MATCHES "/code/godot/tests/")
        set(_EXCL TRUE)
    endif()
    if(_SN MATCHES "/code/godot/test_[^/]+$")
        set(_EXCL TRUE)
    endif()

    # Dead/replaced modules
    if(_SN MATCHES "godot_shader_props\\.cpp$" OR
       _SN MATCHES "godot_surface_effects\\.cpp$" OR
       _SN MATCHES "godot_animation_events\\.cpp$" OR
       _SN MATCHES "godot_animation_event_accessors\\.cpp$" OR
       _SN MATCHES "godot_anim_blend\\.cpp$" OR
       _SN MATCHES "godot_speaker_entities\\.cpp$")
        set(_EXCL TRUE)
    endif()

    # Client sound/backend exclusions (replaced by godot_sound.cpp)
    if(_SN MATCHES "/code/client/")
        get_filename_component(_BN "${SRC}" NAME)
        set(_EXCLUDED_BASENAMES
            snd_dma.c snd_dma_new.cpp
            snd_mem.c snd_mem_new.cpp
            snd_mix.c snd_main.c
            snd_openal.c snd_openal_new.cpp
            snd_miles_new.cpp
            snd_adpcm.c snd_altivec.c snd_wavelet.c
            snd_codec.c
            snd_codec_mp3.c snd_codec_ogg.c snd_codec_opus.c snd_codec_wav.c
            snd_info.cpp
            qal.c libmumblelink.c cl_curl.c
            snd_main_new.cpp
        )
        list(FIND _EXCLUDED_BASENAMES "${_BN}" _IDX)
        if(NOT _IDX EQUAL -1)
            set(_EXCL TRUE)
        endif()
    endif()

    if(_EXCL)
        list(APPEND _EXCLUDED "${SRC}")
    endif()
endforeach()

if(_EXCLUDED)
    list(REMOVE_ITEM ALL_SOURCES ${_EXCLUDED})
endif()

# Windows SDK um/ directory: ensures system <windows.h> is found before
# code/fgame/windows.h (which is a game entity class header).
# MSVC searches /I directories in order, so prepending the SDK um/ path
# before code/fgame ensures the system header wins for <windows.h> while
# local #include "windows.h" from within code/fgame/ still resolves via
# MSVC's "search parent directory of including file" rule.
if(MSVC)
    if(DEFINED ENV{WindowsSdkDir} AND DEFINED ENV{WindowsSDKVersion})
        string(REPLACE "\\" "" _SDK_VER "$ENV{WindowsSDKVersion}")
        set(_SDK_UM "$ENV{WindowsSdkDir}include/${_SDK_VER}/um")
        if(EXISTS "${_SDK_UM}")
            target_include_directories(openmohaa PRIVATE "${_SDK_UM}")
        endif()
    endif()
endif()
target_include_directories(openmohaa PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/code/fgame
)

endif()
